cmake_minimum_required(VERSION 3.20)
project(p0_3_bench CXX)
set(CMAKE_CXX_STANDARD 20)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(S2S_USE_SOLUTIONS "Bench the reference D2 instead of your starter" OFF)
find_package(Threads REQUIRED)
set(S2S_COMMON ${CMAKE_CURRENT_SOURCE_DIR}/../../../common/include)
add_executable(false_sharing_bench false_sharing_bench.cpp)
target_include_directories(false_sharing_bench PRIVATE ${S2S_COMMON})
target_link_libraries(false_sharing_bench Threads::Threads)
if(S2S_USE_SOLUTIONS)
  set(d2 ${CMAKE_CURRENT_SOURCE_DIR}/../exercises/solutions/04-kv-block-allocator)
else()
  set(d2 ${CMAKE_CURRENT_SOURCE_DIR}/../exercises/04-kv-block-allocator)
endif()
add_executable(allocator_bench allocator_bench.cpp)
target_include_directories(allocator_bench PRIVATE ${S2S_COMMON} ${d2})
target_link_libraries(allocator_bench Threads::Threads)
