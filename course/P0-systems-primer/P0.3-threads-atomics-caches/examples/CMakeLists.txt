cmake_minimum_required(VERSION 3.20)
project(p0_3_examples C CXX)
set(CMAKE_CXX_STANDARD 20)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Threads REQUIRED)
set(S2S_COMMON ${CMAKE_CURRENT_SOURCE_DIR}/../../../common/include)
add_executable(01_proc_vs_thread 01_proc_vs_thread.c)
target_link_libraries(01_proc_vs_thread Threads::Threads)
foreach(t 02_false_sharing 03_atomics 04_thread_pool)
  add_executable(${t} ${t}.cpp)
  target_include_directories(${t} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${S2S_COMMON})
  target_link_libraries(${t} Threads::Threads)
endforeach()
