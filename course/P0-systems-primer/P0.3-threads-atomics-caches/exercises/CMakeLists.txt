# P0.3 exercises.  cmake -S . -B ../build/ex && cmake --build ../build/ex -j && ctest --test-dir ../build/ex --output-on-failure
# Add -DS2S_TSAN=ON to build everything under ThreadSanitizer (recommended for 01, 04 and 05).
cmake_minimum_required(VERSION 3.20)
project(p0_3_exercises CXX)
set(CMAKE_CXX_STANDARD 20)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
option(S2S_USE_SOLUTIONS "Build tests against the reference solutions" OFF)
option(S2S_TSAN "Build with ThreadSanitizer" OFF)
if(S2S_TSAN)
  add_compile_options(-fsanitize=thread -g -O1)
  add_link_options(-fsanitize=thread)
endif()
find_package(Threads REQUIRED)
enable_testing()
set(S2S_COMMON ${CMAKE_CURRENT_SOURCE_DIR}/../../../common/include)
function(s2s_exercise name)
  if(S2S_USE_SOLUTIONS)
    set(impl_dir ${CMAKE_CURRENT_SOURCE_DIR}/solutions/${name})
  else()
    set(impl_dir ${CMAKE_CURRENT_SOURCE_DIR}/${name})
  endif()
  add_executable(${name} ${CMAKE_CURRENT_SOURCE_DIR}/${name}/test.cpp)
  target_include_directories(${name} PRIVATE ${impl_dir} ${S2S_COMMON} ${CMAKE_CURRENT_SOURCE_DIR}/../examples)
  target_link_libraries(${name} Threads::Threads)
  add_test(NAME ${name} COMMAND ${name})
endfunction()
foreach(e 01-fix-the-race 02-pad-counters 03-thread-pool 04-kv-block-allocator 05-spsc-queue)
  s2s_exercise(${e})
endforeach()
