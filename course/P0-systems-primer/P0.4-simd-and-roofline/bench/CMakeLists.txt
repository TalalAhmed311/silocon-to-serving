cmake_minimum_required(VERSION 3.20)
project(p0_4_bench CXX)
set(CMAKE_CXX_STANDARD 20)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-O3 -march=native)
option(S2S_USE_SOLUTIONS "Bench the reference D1 instead of your starter" OFF)
find_package(Threads REQUIRED)
if(S2S_USE_SOLUTIONS)
  set(d1 ${CMAKE_CURRENT_SOURCE_DIR}/../exercises/solutions/03-sgemm-ladder ${CMAKE_CURRENT_SOURCE_DIR}/../exercises/solutions/04-sgemm-threads)
else()
  set(d1 ${CMAKE_CURRENT_SOURCE_DIR}/../exercises/03-sgemm-ladder ${CMAKE_CURRENT_SOURCE_DIR}/../exercises/04-sgemm-threads)
endif()
add_executable(sgemm_bench sgemm_bench.cpp)
target_include_directories(sgemm_bench PRIVATE ${d1} ${CMAKE_CURRENT_SOURCE_DIR}/../examples
  ${CMAKE_CURRENT_SOURCE_DIR}/../../../common/include ${CMAKE_CURRENT_SOURCE_DIR}/../../P0.3-threads-atomics-caches/examples)
target_link_libraries(sgemm_bench Threads::Threads)
