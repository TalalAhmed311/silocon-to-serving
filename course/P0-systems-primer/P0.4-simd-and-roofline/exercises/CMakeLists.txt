# P0.4 exercises.  cmake -S . -B ../build/ex && cmake --build ../build/ex -j && ctest --test-dir ../build/ex --output-on-failure
cmake_minimum_required(VERSION 3.20)
project(p0_4_exercises CXX)
set(CMAKE_CXX_STANDARD 20)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-O3 -march=native)
option(S2S_USE_SOLUTIONS "Build tests against the reference solutions" OFF)
find_package(Threads REQUIRED)
enable_testing()
set(S2S_COMMON ${CMAKE_CURRENT_SOURCE_DIR}/../../../common/include)
set(EXAMPLES ${CMAKE_CURRENT_SOURCE_DIR}/../examples)
set(POOL ${CMAKE_CURRENT_SOURCE_DIR}/../../P0.3-threads-atomics-caches/examples)
function(s2s_exercise name)
  if(S2S_USE_SOLUTIONS)
    set(impl_dir ${CMAKE_CURRENT_SOURCE_DIR}/solutions/${name})
  else()
    set(impl_dir ${CMAKE_CURRENT_SOURCE_DIR}/${name})
  endif()
  add_executable(${name} ${CMAKE_CURRENT_SOURCE_DIR}/${name}/test.cpp)
  # 04 reuses 03's kernels: put 03's directory (starter or solution) on the path too.
  if(S2S_USE_SOLUTIONS)
    set(d1 ${CMAKE_CURRENT_SOURCE_DIR}/solutions/03-sgemm-ladder)
  else()
    set(d1 ${CMAKE_CURRENT_SOURCE_DIR}/03-sgemm-ladder)
  endif()
  target_include_directories(${name} PRIVATE ${impl_dir} ${d1} ${S2S_COMMON} ${EXAMPLES} ${POOL})
  target_link_libraries(${name} Threads::Threads)
  add_test(NAME ${name} COMMAND ${name})
endfunction()
foreach(e 01-saxpy 02-hsum 03-sgemm-ladder 04-sgemm-threads 05-int8-gemv)
  s2s_exercise(${e})
endforeach()
