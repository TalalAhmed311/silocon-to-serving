# P0.5 exercises: tests build against YOUR ops/ops.hpp (or the reference with -DS2S_USE_SOLUTIONS=ON).
#   uv run python make_fixtures.py ../../../../build/p05-fixtures      (from this folder; or pass any dir and -DFIXTURES=)
#   cmake -S . -B ../../../../build/p05-ex && cmake --build ../../../../build/p05-ex -j && ctest --test-dir ../../../../build/p05-ex --output-on-failure
cmake_minimum_required(VERSION 3.20)
project(p0_5_exercises CXX)
set(CMAKE_CXX_STANDARD 20)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-O2 -march=native)
option(S2S_USE_SOLUTIONS "Build against platform/engine/v0/src/ops.hpp" OFF)
set(V0 ${CMAKE_CURRENT_SOURCE_DIR}/../../../../platform/engine/v0/src)
set(FIXTURES ${CMAKE_CURRENT_SOURCE_DIR}/../../../../build/p05-fixtures CACHE PATH "make_fixtures.py output")
if(S2S_USE_SOLUTIONS)
  set(OPS ${V0})
  set(Q8 ${CMAKE_CURRENT_SOURCE_DIR}/solutions)
else()
  set(OPS ${CMAKE_CURRENT_SOURCE_DIR}/ops)
  set(Q8 ${CMAKE_CURRENT_SOURCE_DIR}/05-int8)
endif()
find_package(Threads REQUIRED)
enable_testing()
foreach(t 01_rmsnorm_rope 02_attention 03_full_forward 04_top_p 05_q8_matvec)
  add_executable(${t} tests/${t}.cpp)
  # BEFORE: the chosen ops.hpp wins over platform/engine/v0/src/ops.hpp
  target_include_directories(${t} BEFORE PRIVATE ${OPS} ${Q8})
  target_include_directories(${t} PRIVATE ${V0} ${CMAKE_CURRENT_SOURCE_DIR}/../../../common/include)
  target_link_libraries(${t} Threads::Threads)
  add_test(NAME ${t} COMMAND ${t} ${FIXTURES})
endforeach()
# Your engine binary (same main.cpp as the platform, your ops): ./s2s-engine-ex --model .../tiny ...
add_executable(s2s-engine-ex ${V0}/main.cpp)
target_include_directories(s2s-engine-ex BEFORE PRIVATE ${OPS})
target_include_directories(s2s-engine-ex PRIVATE ${V0})
target_link_libraries(s2s-engine-ex Threads::Threads)
