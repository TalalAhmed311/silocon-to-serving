# Lane B local harness: builds every laneB-cuda/L*/solutions/*/test.cu into a test, plus benches.
#   cmake -S laneB-cuda -B build/laneB -DCMAKE_BUILD_TYPE=Release   (-DCMAKE_CUDA_ARCHITECTURES=native, or e.g. 75 for T4)
#   cmake --build build/laneB -j && ctest --test-dir build/laneB --output-on-failure
# Each test also accepts --bench to print a timing table (see harness/README.md).
# Requires: CUDA 12.4+, CMake 3.24+, an NVIDIA GPU (T2). The cuBLAS/CUB baselines link CUDA::cublas.
cmake_minimum_required(VERSION 3.24)
project(laneB LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES native)
endif()
find_package(CUDAToolkit REQUIRED)
enable_testing()
set(S2S_COMMON ${CMAKE_CURRENT_SOURCE_DIR}/../course/common/include)
file(GLOB problem_tests CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/L*/solutions/*/test.cu)
foreach(t ${problem_tests})
  get_filename_component(dir ${t} DIRECTORY)
  get_filename_component(name ${dir} NAME)
  get_filename_component(level_dir ${dir}/../.. ABSOLUTE)
  get_filename_component(level ${level_dir} NAME)
  string(SUBSTRING ${level} 0 2 lvl)
  set(target ${lvl}_${name})
  add_executable(${target} ${t})
  target_include_directories(${target} PRIVATE ${dir} ${CMAKE_CURRENT_SOURCE_DIR}/harness/include ${S2S_COMMON})
  target_link_libraries(${target} PRIVATE CUDA::cudart CUDA::cublas)
  target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr -lineinfo>)
  add_test(NAME ${target} COMMAND ${target})
endforeach()
